function initLanguageDropdown() {

  const langMap = {
    '/':  'English',
    'de': 'Deutsch',
    'jp': '日本語',
    'es': 'Español',
    'fr': 'Français',
  };

  function getLangKey() {
    const segment = window.location.pathname.split('/').filter(Boolean)[0];
    return segment && langMap[segment] ? segment : '/';
  }

  function findMatchingItem(items, langKey) {
    let selectedItem = null;
    items.forEach(li => {
      const a = li.querySelector('a');
      if (langKey === '/') {
        if (!a) selectedItem = li;
      } else {
        if (a && a.getAttribute('href') === `/${langKey}/`) selectedItem = li;
      }
    });
    return selectedItem;
  }

  function updateDropdown(nav) {
    if (!nav) return;

    const list = nav.querySelector('.languages-list');
    const trigger = nav.querySelector('.nav-top-drop');
    if (!list || !trigger) return;

    const langKey = getLangKey();
    const label = langMap[langKey];

    const selectedItem = findMatchingItem(list.querySelectorAll('li'), langKey);

    if (selectedItem) list.prepend(selectedItem);

    const icon = trigger.querySelector('.icon');
    trigger.textContent = '';
    if (icon) trigger.appendChild(icon);
    trigger.append(' ' + label);
  }

  // Run for both occurrences
  document.querySelectorAll('li.languages').forEach(updateDropdown);
}
