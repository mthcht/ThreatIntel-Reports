<script>
(function () {
  if (window.location.pathname.indexOf('/guides/') !== 0) return;
  var contents = document.querySelector('[fs-toc-element="contents"]');
  var toc = document.querySelector('.tableofcontent_wrapper');
  if (!contents || !toc) return;
  var block = toc.closest('.margin-bottom') || toc;
  var h2s = contents.querySelectorAll('h2');
  if (h2s.length > 1) {
    h2s[1].parentNode.insertBefore(block, h2s[1]);
  } else {
    var firstP = contents.querySelector('p');
    if (firstP && firstP.parentNode) firstP.parentNode.insertBefore(block, firstP.nextSibling);
  }
})();
</script></body></html>
