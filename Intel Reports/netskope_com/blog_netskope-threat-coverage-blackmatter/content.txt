  (function () {
    let pagefind;
    let debounceTimer;

    async function loadPagefind() {
      if (pagefind) return pagefind;
      try {
        pagefind = await import('/pagefind/pagefind.js');
        await pagefind.init();
      } catch {
        pagefind = null;
      }
      return pagefind;
    }

    function initSearch() {
      const overlay = document.getElementById('search-overlay');
      const openBtn = document.getElementById('search-open');
      const closeBtn = document.getElementById('search-close');
      const input = document.getElementById('search-input');
      const results = document.getElementById('search-results');

      function openModal() {
        overlay?.classList.remove('hidden');
        overlay?.classList.add('flex');
        input?.focus();
        loadPagefind();
      }

      function closeModal() {
        overlay?.classList.add('hidden');
        overlay?.classList.remove('flex');
        if (input) input.value = '';
        if (results) results.innerHTML = '';
      }

      openBtn?.addEventListener('click', openModal);
      closeBtn?.addEventListener('click', closeModal);
      overlay?.addEventListener('click', (e) => {
        if (e.target === overlay) closeModal();
      });
      document.addEventListener('keydown', (e) => {
        if (e.key === 'Escape') closeModal();
        if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
          e.preventDefault();
          openModal();
        }
      });

      input?.addEventListener('input', () => {
        clearTimeout(debounceTimer);
        debounceTimer = setTimeout(runSearch, 200);
      });

      async function runSearch() {
        const query = input?.value.trim() ?? '';
        if (!results) return;
        if (!query) {
          results.innerHTML = '';
          return;
        }

        const pf = await loadPagefind();
        if (!pf) {
          results.innerHTML = `<p class="p-4 text-sm text-(--text-secondary)">Search is unavailable in development. It runs against the built site.</p>`;
          return;
        }

        const search = await pf.search(query);
        if (search.results.length === 0) {
          results.innerHTML = `<p class="p-4 text-sm text-(--text-secondary)">No results for "${query}".</p>`;
          return;
        }

        const items = await Promise.all(search.results.slice(0, 8).map((r) => r.data()));
        results.innerHTML = items
          .map(
            (item) => `
            <a href="${item.url}" class="block px-4 py-3 border-b border-(--border-subtle) last:border-0 hover:bg-(--bg-tertiary) transition-colors">
              <p class="font-medium text-(--text-primary) text-sm mb-1">${item.meta?.title ?? item.url}</p>
              <p class="text-xs text-(--text-secondary) line-clamp-2">${item.excerpt}</p>
            </a>
          `,
          )
          .join('');
      }
    }

    // Initialize on page load and after Astro view transitions
    document.addEventListener('astro:page-load', initSearch);
  })();

lg:px-8 py-12 bg-(--bg-primary)"><div class="prose max-w-none"><h2 id="summary">Summary</h2>
<p>In July of 2021, a new ransomware named <a href="https://malpedia.caad.fkie.fraunhofer.de/details/win.blackmatter">BlackMatter</a> emerged and <a href="https://therecord.media/blackmatter-ransomware-targets-companies-with-revenues-of-100-million-and-more/">was being advertised in web forums</a> where the group was searching for compromised networks from companies with revenues of $100 million or more per year. Although they are not advertising as a Ransomware-as-a-Service (RaaS), the fact they are looking for “partners” is an indication that they are operating in this model. Furthermore, the group is claiming to have combined features from larger groups, such as <a href="/library/malware/darkside" class="threat-link">DarkSide</a> and <a href="/library/malware/revil" class="threat-link">REvil</a> (a.k.a. <a href="/library/malware/revil" class="threat-link">Sodinokibi</a>).</p>
<p><img src="/_astro/tc-blackmatter-1.BoB6RBqw_27BJ7o.webp" alt="Screenshot of BlackMatter advertisement in a web forum." loading="lazy" decoding="async" width="1000" height="864"></p>
<p><a href="/library/malware/blackmatter" class="threat-link">BlackMatter</a> advertisement in a web forum. (Source: <a href="https://therecord.media/blackmatter-ransomware-targets-companies-with-revenues-of-100-million-and-more/">The Record</a>)</p>
<p>According to an <a href="https://therecord.media/an-interview-with-blackmatter-a-new-ransomware-group-thats-learning-from-the-mistakes-of-darkside-and-revil/">interview</a> with an alleged representative from BlackMatter, they have incorporated the ideas of <a href="/blog/2021/08/netskope-threat-coverage-lockbit">LockBit</a>, <a href="/blog/2021/07/netskope-threat-coverage-revil">REvil</a>, and <a href="/blog/2021/05/netskope-threat-coverage-darkside">DarkSide</a>, after studying their ransomware in detail. Also, the BlackMatter representative believes that other ransomware groups have disappeared from the scene due to attention from governments following high-profile attacks.  BlackMatter plans to avoid such attention by being careful not to infect any critical infrastructure. This is echoed on their website, which states they are not willing to attack hospitals, critical infrastructures, defense industry, and non-profit companies.</p>
<p><img src="/_astro/tc-blackmatter-2-1024x423.ClizBoqp_ZLQdt0.webp" alt="Screenshot of Main page of BlackMatter’s website, hosted on the deep web." loading="lazy" decoding="async" width="1024" height="423"></p>
<p>Main page of BlackMatter’s website, hosted on the deep web.</p>
<p>The oil and gas industry is also excluded from the target list, a reference to the Colonial Pipeline attack where DarkSide <a href="https://www.washingtonpost.com/business/2021/05/08/cyber-attack-colonial-pipeline/">stopped the fuel delivery across the Southeastern of the United States</a>, followed by the <a href="https://www.nytimes.com/2021/05/14/business/darkside-pipeline-hack.html">shut down of the ransomware operation</a> due to the pressure from law enforcement. The BlackMatter spokesperson also said that the Colonial PIpeline attack was a key factor for the shutdown of REvil and DarkSide, and that’s why they are excluding this kind of sector from the target list.</p>
<p>BlackMatter already claims to have hit three victims, each listed on their deep web site, which follows the same standard from other groups, containing the name of the attacked company, a summary of what data they have stolen, and the deadline for the ransom before the data is published.</p>
<p><img src="/_astro/tc-blackmatter-3.MgRyU4B1_ZXLCnX.webp" alt="Screenshot showing one of the DarkSide targets, with leaked data on the website." loading="lazy" decoding="async" width="386" height="495"></p>
<p>One of the DarkSide targets, with leaked data on the website.</p>
<p>One of the companies infected by BlackMatter is <a href="https://www.cisoadvisor.com.br/anunciado-ataque-a-fabricante-de-coca-cola-no-brasil/">SolarBR</a>, which is the second-largest manufacturer of Coca-Cola in Brazil, where the group claimed to have stolen 50 GB of confidential finance, logistics, development, and other data.</p>

<h2 id="threat">Threat</h2>
<p>Like other malware, BlackMatter implements many techniques to avoid detection and make reverse engineering more challenging. The first item we would like to cover is how BlackMatter dynamically resolves API calls to hide them from the PE import table.</p>
<p>This is done by a multi-step process. First, the malware creates a unique hash that will identify both the DLL and API name that needs to be executed. To make this a bit harder for static detections, the real hash value is encrypted with a simple XOR operation. In this case, the key is <strong>0x22065FED</strong>.</p>
<p><img src="/_astro/tc-blackmatter-6.CZdB3g3k_26zK0w.webp" alt="Figure showing function that loads the import based on a hash." loading="lazy" decoding="async" width="361" height="167"></p>
<p>Figure 1. Function that loads the import based on a hash.</p>
<p>In the example above, after the XOR operation, the value <strong>0x27D05EB2</strong> is passed as a parameter to the function responsible for searching and loading the API. The code first enumerates all the DLLs that are loaded within the process through a common but interesting technique.</p>
<p>First, it loads the <a href="https://docs.microsoft.com/en-us/windows/win32/api/winternl/ns-winternl-peb">Process Environment Block (PEB)</a> address, which is located in the <a href="https://docs.microsoft.com/en-us/windows/win32/api/winternl/ns-winternl-teb">Thread Environment Block (TEB)</a>. Then, it loads the doubly linked list that contains all the loaded modules for the process, located in the <a href="https://docs.microsoft.com/en-us/windows/win32/api/winternl/ns-winternl-peb_ldr_data">PEB_LDR_DATA</a> structure.</p>
<p><img src="/_astro/tc-blackmatter-7.C1F22_mA_2n4ddD.webp" alt="Figure showing BlackMatter function searching loaded modules using the PEB." loading="lazy" decoding="async" width="591" height="174"></p>
<p>Figure 2. BlackMatter function searching loaded modules using the PEB.</p>
<p>Once the loaded DLL is located, the function retrieves the DLL’s offset, finds the PE header address, and then calculates the offset of the PE export directory, so it can enumerate the APIs exported by the DLL.</p>
<p>If the export table is found, the ransomware then calculates the hash value for both DLL and API name, using the following function:</p>
<p><img src="/_astro/tc-blackmatter-8.BkG0PmlC_Zmi2iU.webp" alt="Figure showing the function used by BlackMatter to calculate the hash of the string." loading="lazy" decoding="async" width="596" height="326"></p>
<p>Figure 3. Function used by BlackMatter to calculate the hash of the string.</p>
<p>To get the unique hash, the ransomware first calculates the hash only for the DLL name.</p>
<p><img src="/_astro/tc-blackmatter-9.hjqaShLl_1qU3bn.webp" alt="Figure showing hash generation for the DLL “kernel32.dll”" loading="lazy" decoding="async" width="537" height="170"></p>
<p>Figure 4. Hash generation for the DLL “kernel32.dll”</p>
<p>In the example above, the hash for the DLL “<code>kernel32.dll</code>” is <strong>0xB1FC7F66</strong>, which is then used by this same function to calculate the hash of the API name.</p>
<p><img src="/_astro/tc-blackmatter-10.B6PLNFFD_2389mb.webp" alt="Figure generating the final hash for DLL + API name" loading="lazy" decoding="async" width="484" height="163"></p>
<p>Figure 5. Generating the final hash for DLL + API name</p>
<p>Therefore, using the same function again, the malware has generated the hash <strong>0x27D05EB2</strong> for the DLL “<code>kernel32.dll</code>” and the API “<code>LoadLibraryA</code>”, which is exactly the same value the malware is seeking, as demonstrated in Figure 1.</p>
<p>If the hash generated by the function matches the hash the malware passed as a parameter, the offset for the API is stored in memory, so the function can be called.</p>
<p><img src="/_astro/tc-blackmatter-11.BGMvNIW4_Z180h7N.webp" alt="Figure showing BlackMatter’s code before and after the APIs were dynamically resolved." loading="lazy" decoding="async" width="657" height="146"></p>
<p>Figure 6. BlackMatter’s code before and after the APIs were dynamically resolved.</p>
<p>Several DLLs are loaded by BlackMatter dynamically after the executable is running, as we can see below.</p>
<p><img src="/_astro/tc-blackmatter-12-1024x410.BtHaEpI-_23Vn9O.webp" alt="Figure showing DLLs dynamically loaded by BlackMatter." loading="lazy" decoding="async" width="1024" height="410"></p>
<p>Figure 7. DLLs dynamically loaded by BlackMatter.</p>
<p>To make the analysis faster, we’ve created a script that implements the same logic used by BlackMatter for the hash generation. Therefore, the script can be used to locate calls to specific APIs across BlackMatter’s code.</p>
<p><img src="/_astro/tc-blackmatter-13.BdsD-eEp_ZNDRvi.webp" alt="Figure showing script to generate the hash based on the API call." loading="lazy" decoding="async" width="513" height="331"></p>
<p>Figure 8. Script to generate the hash based on the API call.</p>
<p>Another technique used by BlackMatter to stay under the radar is to encrypt all its important strings. In the samples we’ve analyzed, the ransomware used the same key as the one used to generate the hashes for the API loading process.</p>
<p><img src="/_astro/tc-blackmatter-14.mX93n9a-_2w0Mv.webp" alt="Figure showing BlackMatter’s routine for string decryption." loading="lazy" decoding="async" width="416" height="629"></p>
<p>Figure 9. BlackMatter’s routine for string decryption.</p>
<p>After the bytes are organized in memory, the code decrypts the data in 4-byte blocks, using a simple XOR operation with the key <strong>0x22065FED.</strong></p>
<p><img src="/_astro/tc-blackmatter-15.BYRgT6hj_ZWVuAv.webp" alt="Figure showing example of a string decrypted by BlackMatter." loading="lazy" decoding="async" width="635" height="205"></p>
<p>Figure 10. Example of a string decrypted by BlackMatter.</p>
<p>We can find useful information across the decrypted strings, such as registry keys, file names, and others. The full list of decrypted strings can be found in our <a href="https://github.com/netskopeoss/NetskopeThreatLabsIOCs/tree/main/BlackMatter/IOCs">GitHub repository</a>.</p>
<p><img src="/_astro/tc-blackmatter-16.iGkex8qX_1pUrTy.webp" alt="Figure showing some of BlackMatter’s decrypted strings." loading="lazy" decoding="async" width="478" height="200"></p>
<p>Figure 11. Some of BlackMatter’s decrypted strings.</p>
<p>BlackMatter also has an encrypted configuration inside the binary, located in a fake PE resource section.</p>
<p><img src="/_astro/tc-blackmatter-17.DK-nCOh__Z1tBtry.webp" alt="Figure showing BlackMatter’s encrypted configuration." loading="lazy" decoding="async" width="587" height="285"></p>
<p>Figure 12. BlackMatter’s encrypted configuration.</p>
<p>The first 4 bytes in the section are the initial decryption key, the following 4 bytes represent the size of the data, and the rest of the bytes are the encrypted configuration. The data is then decrypted using a rolling XOR algorithm.</p>
<p>A new decryption key is generated every 4 bytes, using a dynamic seed and a constant, which is <strong>0x8088405</strong> in all the samples we have analyzed so far.</p>
<p><img src="/_astro/tc-blackmatter-18.D2Reeq01_Z3UANV.webp" alt="Figure showing the stub that generates the decryption key." loading="lazy" decoding="async" width="441" height="254"></p>
<p>Figure 13. Stub that generates the decryption key.</p>
<p>The decrypted configuration is compressed using <a href="https://ibsensoftware.com/products_aPLib.html">aPLib</a>, so we need to decompress the bytes to get the information. Once this process is done, we can read the contents of the configuration. At the beginning, we can find the attacker’s RSA public key, the AES key used to encrypt C2 communication, as well as a 16-byte value named “<code>bot_company</code>”.</p>
<p><img src="/_astro/tc-blackmatter-19.CUdrppat_wT66O.webp" alt="Figure showing BlackMatter’s decrypted configuration." loading="lazy" decoding="async" width="640" height="463"></p>
<p>Figure 14. BlackMatter’s decrypted configuration.</p>
<p>Aside from that, the configuration also includes several base64 encoded strings that contain sensitive strings used by the malware, like the C2 server addresses.</p>
<p><img src="/_astro/tc-blackmatter-20.Cz9A6e-d_1PNYLx.webp" alt="Figure decoding BlackMatter’s C2 server addresses." loading="lazy" decoding="async" width="744" height="222"></p>
<p>Figure 15. Decoding BlackMatter’s C2 server addresses.</p>
<p>Among the strings, there is also a list of processes and services that the ransomware attempts to stop \ terminate.</p>
<p><img src="/_astro/tc-blackmatter-21.vpYnbPsQ_ZwwQC2.webp" alt="Figure showing ransomware trying to open the VSS service." loading="lazy" decoding="async" width="356" height="122"></p>
<p>Figure 16. Ransomware trying to open the VSS service.</p>
<p>To speed up the analysis, we have <a href="https://github.com/netskopeoss/NetskopeThreatLabsIOCs/blob/main/BlackMatter/Scripts/decrypt_data.py">created a script</a> that is able to decrypt the strings and the configuration from BlackMatter samples.</p>
<p><img src="/_astro/tc-blackmatter-22.BtlDGj_6_f4pGq.webp" alt="Figure decrypting BlackMatter’s strings." loading="lazy" decoding="async" width="775" height="422"></p>
<p>Figure 17. Decrypting BlackMatter’s strings.</p>
<p>The script also decodes all base64 values from the configuration automatically:</p>
<p><img src="/_astro/tc-blackmatter-23.4_EPiRBM_Z203wlN.webp" alt="Figure showing BlackMatter’s C2 server addresses." loading="lazy" decoding="async" width="379" height="106"></p>
<p>Figure 18. BlackMatter’s C2 server addresses.</p>
<p>BlackMatter communicates with the C2 server in order to send information to the attackers. It first loads a JSON structure in memory, containing all the information that will be sent.</p>
<p><img src="/_astro/tc-blackmatter-24.TKnfRS3B_oQIFQ.webp" alt="Figure showing information that will be sent to the C2 address." loading="lazy" decoding="async" width="523" height="242"></p>
<p>Figure 19. Information that will be sent to the C2 address.</p>
<p>Prior to the POST request, the information is encrypted using AES-128 ECB, with the key extracted from the configuration, and then encoded with base64.</p>
<p><img src="/_astro/tc-blackmatter-25.bW4AOIvf_1BocpW.webp" alt="Figure showing BlackMatter sending request to the C2 server." loading="lazy" decoding="async" width="749" height="336"></p>
<p>Figure 20. BlackMatter sending request to the C2 server.</p>
<p>It’s possible to decrypt this information by decoding the base64 and decrypting the data using the key from the configuration file.</p>
<p><img src="/_astro/tc-blackmatter-26.q4USQ9Qg_28fSme.webp" alt="Figure decrypting BlackMatter’s C2 request." loading="lazy" decoding="async" width="810" height="410"></p>
<p>Figure 21. Decrypting BlackMatter’s C2 request.</p>
<p>BlackMatter sends two requests, the first one contains details about the infected environment, and the second one contains details about the encryption process, such as how many files failed to encrypt, the start and end time, etc.</p>
<p>Finally, once the encryption process is complete, the ransom note is created in the same places where there are encrypted files.</p>
<p><img src="/_astro/tc-blackmatter-27-1024x472.Bzni9_rW_Z1u8rAa.webp" alt="Figure showing BlackMatter’s ransom note." loading="lazy" decoding="async" width="1024" height="472"></p>
<p>Figure 22. BlackMatter’s ransom note.</p>
<p>BlackMatter changes the background image, a common practice among ransomware creators.</p>
<p><img src="/_astro/tc-blackmatter-28-1024x233.CwrgO24d_14mMKx.webp" alt="Figure showing BlackMatter’s custom background" loading="lazy" decoding="async" width="1024" height="233"></p>
