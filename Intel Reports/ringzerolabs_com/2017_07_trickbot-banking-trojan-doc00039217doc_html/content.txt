(function() {
    const canvas = document.getElementById('kernel-canvas');
    const ctx = canvas.getContext('2d');
    const container = document.getElementById('bannerContainer');
    let width, height, particles = [], mouse = { x: -10000, y: -10000 };
    const MAX_HOVER_COUNT = 3;
    const CAPTURE_RADIUS = 70;
    const RELEASE_RADIUS = 100;
    let lockedParticles = [];

    const symbols = [
        'NtWriteVirtualMemory', 'xor eax, eax', '0x41414141', 'KeStackAttachProcess',
        'LdrLoadDll', 'ZwProtectVirtualMemory', 'SetWindowsHookEx', 'Shellcode_Run',
        'sub_140007A10', 'hook_dispatch', '0xDEADC0DE', 'MSR_LSTAR_WRITE',
        'IofCallDriver_Hook', 'ObfReferenceObject_Proxy', '0x9090909090',

        'mov rax, [rcx+18h]', 'push rbp', 'lea rdi, [rsp+20h]', 'test eax, eax',
        'jnz short loc_14000', 'call qword ptr [rip+210Ah]', 'add esp, 0Ch',
        'repne scasb', 'cmp byte ptr [rax], 0', 'jmp r12', 'sysret', 'cli', 'hlt',
        'mov r15, r8', 'pop r12', 'shl rdx, 4', 'and eax, 0FFFFh', 'movups xmm1, [rbx]',
        'void* __fastcall mem_alloc(size_t sz)', 'if (status < 0) return;',
        'PsSetCreateProcessNotifyRoutine', 'while (i < buffer_len)',
        'NTSTATUS NtQuerySystemInformation', 'DeviceIoControl', 'memcpy(dst, src, len)',
        'nt!ExAllocatePoolWithTag', 'hal!HalRequestIpi', 'PsGetCurrentProcess',
        'KeAcquireSpinLock(&lock, &irql)', 'IoCreateDevice(DriverObject, ...)',
        'struct _EPROCESS* TargetProcess', 'KTHREAD.ServiceTable', 'CONTAINING_RECORD(...)',
        'MmIsAddressValid(ptr)', 'ExFreePoolWithTag(tag_RZL)', 'RtlInitUnicodeString(&str, L"\\??\\")'
    ];
    const suspicious = ['NtWrite', 'xor', '0x', 'Attach', 'Shell', 'Protect', 'Hook', 'Ldr', 'DEADC0DE', 'Alloc', 'Privilege', 'MSR'];

    function init() {
        width = canvas.width = container.offsetWidth;
        height = canvas.height = container.offsetHeight;
        particles = [];
        for (let i = 0; i < Math.floor(width / 22); i++) {
            const p = new Particle();
            p.x = Math.random() * (width + 800);
            particles.push(p);
        }
    }

    class Particle {
        constructor() { this.reset(true); }
        reset(isInitial) {
            if (!isInitial) this.x = width + (Math.random() * 600);
            this.y = 15 + Math.random() * (height - 30);
            this.baseY = this.y;
            this.speed = 0.08 + Math.random() * 0.4;
            this.text = symbols[Math.floor(Math.random() * symbols.length)];
            this.isSuspicious = suspicious.some(key => this.text.includes(key));
            this.opacity = 0.08 + Math.random() * 0.20;
            this.active = false;
            this.dist = 9999;
        }
        update() {
            this.dist = Math.hypot(this.x - mouse.x, this.y - mouse.y);
            if (!this.active) {
                this.x -= this.speed;
                this.y += (this.baseY - this.y) * 0.05;
            } else { 
                this.x -= 0.015; 
            }
            if (this.x < -600) this.reset(false);
        }
        draw() {
            ctx.font = this.active ? "bold 11px Roboto" : "10px Roboto";
            ctx.fillStyle = this.isSuspicious ? (this.active ? '#ff3333' : `rgba(255, 51, 51, ${this.opacity + 0.1})`) : (this.active ? '#00ff41' : `rgba(0, 255, 65, ${this.opacity})`);
            ctx.fillText(this.text, this.x, this.y);
            if (this.active && mouse.x > 0) {
                ctx.strokeStyle = this.isSuspicious ? 'rgba(255, 51, 51, 0.25)' : 'rgba(0, 255, 65, 0.15)';
                ctx.beginPath(); 
                ctx.moveTo(this.x, this.y - 4); 
                ctx.lineTo(mouse.x, mouse.y); 
                ctx.stroke();
            }
        }
    }

    function animate() {
        ctx.fillStyle = '#050505';
        ctx.fillRect(0, 0, width, height);

        // Filter out particles that are too far away
        lockedParticles = lockedParticles.filter(p => p.dist < RELEASE_RADIUS);

        if (lockedParticles.length < MAX_HOVER_COUNT && mouse.x > -5000) {
            const candidates = particles.filter(p => !p.active && p.dist < CAPTURE_RADIUS).sort((a, b) => a.dist - b.dist);
            while (lockedParticles.length < MAX_HOVER_COUNT && candidates.length > 0) {
                const p = candidates.shift(); 
                p.active = true; 
                lockedParticles.push(p);
            }
        }

        particles.forEach(p => { 
            p.active = lockedParticles.includes(p); 
            p.update(); 
            p.draw(); 
        });
        requestAnimationFrame(animate);
    }

    container.addEventListener('mousemove', e => {
        const rect = container.getBoundingClientRect();
        mouse.x = e.clientX - rect.left; 
        mouse.y = e.clientY - rect.top;
    });

    container.addEventListener('mouseleave', () => {
        mouse.x = -10000;
        mouse.y = -10000;
        lockedParticles.forEach(p => p.active = false);
        lockedParticles = [];
    });

    window.addEventListener('resize', init);
    init(); animate();
})();

<h3 class='post-title entry-title' itemprop='name'>
TrickBot Banking Trojan - DOC00039217.doc
</h3>
<div class='post-header'>
<div class='post-header-line-1'></div>
</div>
<div class='post-body entry-content' id='post-body-1442145604602843170' itemprop='articleBody'>
<div>
    <section style="display: flex; align-items: flex-start; gap: 20px; padding-bottom: 20px;">
        <div style="flex-shrink: 0;">
            <img alt="malware analysis trickbot banking trojan" height="100" loading="lazy" src="https://blogger.googleusercontent.com/img/b/R29vZ2xl/AVvXsEjgoPi09MgZR9vfnD51_s-Yxos2OezYpbtVKLlQezJfBO1nm1xzHX4XSlIHZU3Th_f-rcWTZuN3LAO0qUx42EGNplCqi_dWfQgDMbxNWpkJvIz7t7SjPjdFq2rC-WtsaBo_jE0S-AvCJDs/s1600-rw/doczip.png" style="border: 1px solid #757575;" width="150" />
        </div>
        <div style="margin-top: 0;">
            <p style="margin-top: 0;">
                <code>DOC00039217.doc</code> is a malicious Word document that utilizes VBA macros to initiate a multi-stage infection, ultimately deploying the <strong>TrickBot</strong> banking trojan.
            </p>
            <p style="line-height: 1.4;">
                <strong>Filename:</strong> DOC00039217.doc<br />
                <strong>MD5:</strong> 31529e5221e16a522e8aece4998036d7<br />
                <strong>Sample:</strong> <a href="https://www.reverse.it/sample/f934141126cdc13d1078e8c463b7241594d354c5513aa025d32853a658d237e7?environmentId=100" target="_blank">Download via Reverse.it</a>
            </p>
        </div>
    </section>
  <a name="more"></a>

    <br /><br />

    <section>
        <h2>Technical Analysis Walkthrough</h2>
        <div style="position: relative; padding-bottom: 56.25%; height: 0; overflow: hidden; width: 100%; border: 1px solid #ddd;">
            <iframe src="https://www.youtube.com/embed/b6aul3QH0HY" style="position: absolute; top: 0; left: 0; width: 100%; height: 100%;" frameborder="0" allowfullscreen></iframe>
        </div>
    </section>

    <br /><br />

    <section>
        <h2>Stage 1: Document & VBA Analysis</h2>
        <p>
            Initial header analysis reveals "PK" signatures and XML references, confirming this is an Office Open XML (DOCM) file masked as a .DOC. By renaming the extension to .ZIP, we can extract the internal contents.
        </p>
        <center>
            <img loading="lazy" src="https://blogger.googleusercontent.com/img/b/R29vZ2xl/AVvXsEjgoPi09MgZR9vfnD51_s-Yxos2OezYpbtVKLlQezJfBO1nm1xzHX4XSlIHZU3Th_f-rcWTZuN3LAO0qUx42EGNplCqi_dWfQgDMbxNWpkJvIz7t7SjPjdFq2rC-WtsaBo_jE0S-AvCJDs/s1600-rw/doczip.png" style="max-width: 100%; border: 1px solid #eee;" />
        </center>
        <p>
            The <code>vbaProject.bin</code> file contains the primary downloader script. Upon execution, it reaches out to <code>http://appenzeller.fr/aaaa</code> to retrieve the second stage.
        </p>
        <center>
            <img loading="lazy" src="https://blogger.googleusercontent.com/img/b/R29vZ2xl/AVvXsEhRtLTUOl780znkRF6e7Dpy8VCbFJjlY9ez9aabD_klKbl9z_QUw0t3_b-SIah6WI069Hj3sUXL8kJ1MlhI1iPG4IFZvjcAONudiP6V_OidumSc1MwcDVh57UYzLrhXPliRbIAICwElWos/s1600-rw/first_vba.png" style="max-width: 100%;" />
        </center>
    </section>

    <br /><br />

    <section>
        <h2>Stage 2: VBScript & PowerShell Loader</h2>
        <p>
            The file <code>aaaa</code> is a VBScript that leverages <code>Wscript.Shell</code> to invoke PowerShell. It constructs a dynamic URL (<code>amphibiousvehicle.eu/0chb7</code>) to download the final payload.
        </p>
        <center>
            <img loading="lazy" src="https://blogger.googleusercontent.com/img/b/R29vZ2xl/AVvXsEilGBEyKmZJ5pnz6IkQhowqrlenXa6EVoSE1aV-ZMoi5YukM6kGVQJ-aCixqaEw0tYqmeX2P-2H93SBL3Jirve2MToixqfTwfxWFS9Q0Q28c7MjRHoSlutJ2r-Zt-2Wu437E3CvowTJUoA/s1600-rw/second_vba.png" style="max-width: 100%;" />
        </center>
        <p>
            The payload is saved to the <code>%TEMP%</code> folder as <code>petya.exe</code>. Despite the name, this is <strong>not</strong> the Petya ransomware, but the TrickBot trojan.
        </p>
    </section>

    <br /><br />

    <section>
        <h2>Unpacking the Payload (PECompact2)</h2>
        <p>
            The binary is packed with <strong>PECompact2</strong>. To find the Original Entry Point (OEP), we load the file into a debugger and locate the last JMP instruction before the null-byte padding.
        </p>
        <div style="display: flex; flex-direction: column; gap: 10px; align-items: center;">
            <img loading="lazy" src="https://blogger.googleusercontent.com/img/b/R29vZ2xl/AVvXsEi-jYNWv7yOprY8u2GmjE7g9oNM6-G7m1INSuHcyCB1fyK458lCfLCEBfECvK7zrFkwZXa-DNtV7aChenDQREgsQN4YxuCp63iD713hs10YjOFXczEwa1W-PE9Lylv_BZAsfhih1bIzP0Y/s1600-rw/entry.png" />
            <img loading="lazy" src="https://blogger.googleusercontent.com/img/b/R29vZ2xl/AVvXsEjdN6pXhQZzvAlHuiufmJ59XAC6mkC_2x4BG8mpowyjv3PHIVaN-M6iiReZ1KuippmDfn8r5Hob3cxeAhKnef5Z-qeTJaIHmQYIJKLSgNemW96ftQrb1KU5jP2kkRrT55zl56FJHC4VPIE/s1600-rw/step2.png" />
            <img loading="lazy" src="https://blogger.googleusercontent.com/img/b/R29vZ2xl/AVvXsEhtzTRgJ79qi1P8jkLPSNZhPKePvxdg78n0Rt6SgqgArP3P8CAnSIWrOVmkcnwdvdYdWi94PGh5leWCgzteTE6QhRn6eJgslIUwdn9aEMd5B4XHAEDMolrB2GsWWzIZwBT8Mkyv-oM1_GE/s1600-rw/step3.png" />
        </div>
    </section>

    <br /><br />

    <section>
        <h2>Persistence & Process Hollowing</h2>
        <p>
            The malware establishes itself in the <code>%AppData%\Roaming\winapp</code> directory as <code>odsxa.exe</code>. It uses <strong>Process Hollowing</strong> to inject its malicious code into a legitimate <code>svchost.exe</code> process.
        </p>
        <center>
            <img loading="lazy" src="https://blogger.googleusercontent.com/img/b/R29vZ2xl/AVvXsEiVn0U3RKHaxIhdt2BNSZIHfV1erCbo-DK8JyjYXVNPEz3xOwTr3YdN4gfxcEC6ONgAh5S5klDPBl_FzcETozh8LpHaIVCv-IvRRX0UR0HW2oYRTsloF4SDwMwvL03Jk1zWqeDkXBi3-3Y/s1600-rw/directory.png" />
        </center>
        <p>
            This allows the malware to operate within the security context of a trusted system process.
        </p>
        
        <center>
            <img loading="lazy" src="https://blogger.googleusercontent.com/img/b/R29vZ2xl/AVvXsEicavipoLqCfFYplDEtgBokTjmeYslOtP2mGqS6qEDvnaR3g6dYyAqOuXtI3AUU3Yz_-ZxsQ5-lU3afYL1YSu0dw7HkyXq6ULmdf8K0MCtJWkNqtLO0w20MMPp9Qf5LbwpYHitkfZBftMo/s1600-rw/svchost_injection.png" />
        </center>
    </section>

    <br /><br />

    <section>
        <h2>C2 Communication & Modular Payload</h2>
        <p>
            The injected process first retrieves the victim's public IP via <code>ipinfo.io/ip</code>, then begins beaconing to multiple hardcoded C2 IPs over HTTPS.
        </p>
        <center>
            <img loading="lazy" src="https://blogger.googleusercontent.com/img/b/R29vZ2xl/AVvXsEh9no2Db_IP5_hLOf6ZG8VoAWahiz6TNngyoJ3P7sINDbhSUsWqRrLCUrq7lWrV1C68gpDuUv_Mg8fQnLAinVEnZi07fJrotuvQ1_EKOh-rYKyuCo3ewdGAiqjjPHLobrspLlnF2yPWJUM/s1600-rw/get.png" />
        </center>
        <p>
            Over time, the malware downloads encrypted modules into the <code>\modules</code> folder, extending its capability for credential theft and banking fraud.
        </p>
        <div style="display: flex; justify-content: center; gap: 10px; flex-wrap: wrap;">

        <h2>Conclusion & Detection</h2>
        <p>
            This multi-stage campaign highlights the evolution of <strong>TrickBot</strong> as a successor to Dyreza. The use of PowerShell loaders and encrypted modules makes it a highly flexible and dangerous threat.
        </p>
        <p>
            <strong>Best Practices:</strong>
            <ul>
                <li>Block known C2 IPs at the perimeter.</li>
                <li>Disable all Office Macros unless verified by the sender.</li>
                <li>Monitor for suspicious <code>svchost.exe</code> behavior and <code>%AppData%</code> folder modifications.</li>
            </ul>
