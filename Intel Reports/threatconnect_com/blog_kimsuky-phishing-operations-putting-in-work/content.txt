	// populate form fields
	var populateFormFields = function () {
		const form_urls = $('[name="form_urls"]');
		const latest_form_url = $('[name="latest_form_url"]');
		let pathname = location.pathname.replace('/', '');
		if (pathname.endsWith('/')) pathname = pathname.slice(0, -1);

		form_urls.val(';|' + pathname + '|');
		latest_form_url.val(pathname);
	}
	window.addEventListener('message', function (event) {
		if (event.data.eventName === 'onFormReady') {
			$('.hs-form').find('input:text, input[type="email"], input[type="tel"], input[type="number"], textarea, select').not('.processed').each(function () {
				var label = $(this).parents('.hs-form-field').find('label[for]');
				var labelText = label.text().replace('*', ' *');
				if (labelText) {
					$(this).addClass('processed').attr('placeholder', labelText).find('option[disabled]').text(labelText);
				}
				label.addClass('hidden').hide();
			});
			populateFormFields();
			console.log('hs-form ready');
		}
	});
